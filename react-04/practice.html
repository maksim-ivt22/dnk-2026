Мини-проект Study Planner

Задача

Создай приложение для управления учебными задачами.

Приложение должно уметь:
  • получать стартовые задачи с API;
  • показывать их через map();
  • добавлять задачу;
  • удалять задачу;
  • отмечать задачу выполненной;
  • искать задачи;
  • фильтровать задачи.

2. TaskItem.jsx

function TaskItem({
    task,
    toggleTask,
    deleteTask
}) {

    return (

        <div className="task">

            <h3>
                {task.title}
            </h3>


            <p>

                {task.completed
                    ? "Выполнено"
                    : "В процессе"}

            </p>


            <button
                onClick={() =>
                    toggleTask(task.id)
                }
            >

                Изменить статус

            </button>


            <button
                onClick={() =>
                    deleteTask(task.id)
                }
            >

                Удалить

            </button>

        </div>

    );

}


export default TaskItem;

3. TaskForm.jsx

import { useState } from "react";


function TaskForm({ addTask }) {

    const [title, setTitle] = useState("");


    function handleSubmit(event) {

        event.preventDefault();


        if (title.trim() === "") {
            return;
        }


        addTask(title);

        setTitle("");

    }


    return (

        <form onSubmit={handleSubmit}>

            <input
                type="text"
                value={title}
                onChange={(event) =>
                    setTitle(event.target.value)
                }
                placeholder="Новая задача"
            />

            <button type="submit">
                Добавить
            </button>

        </form>

    );

}


export default TaskForm;

4. FilterBar.jsx

function FilterBar({

}) {

    return (

        <div className="filters">

            <button

                }

            </button>


            <button
                onClick={() =>
                    setFilter("active")
                }
            >
                В процессе
            </button>


            <button
                onClick={() =>
                    setFilter("done")
                }
            >
                Выполненные
            </button>

        </div>

    );

}


export default FilterBar;

5. App.jsx

import {
    useEffect,
    useState
} from "react";

import TaskItem
    from "./components/TaskItem";

import TaskForm
    from "./components/TaskForm";

import FilterBar
    from "./components/FilterBar";


function App() {

    const [tasks, setTasks] =
        useState([]);

    const [search, setSearch] =
        useState("");

    const [filter, setFilter] =
        useState("all");


    useEffect(() => {

        async function loadTasks() {

            const response = await fetch(
                "https://jsonplaceholder.typicode.com/todos?_limit=5"
            );

            const data =
                await response.json();

            setTasks(data);

        }


        loadTasks();

    }, []);

6. Добавление

function addTask(title) {

    const newTask = {

        id: Date.now(),

        title: title,

        completed: false

    };


    setTasks([
        ...tasks,
        newTask
    ]);

}

7. Удаление

function deleteTask(id) {

    setTasks(

        tasks.filter(
            task => task.id !== id
        )

    );

}

8. Изменение статуса

function toggleTask(id) {

    setTasks(

        tasks.map(task =>

            task.id === id

                ? {
                    ...task,
                    completed:
                        !task.completed
                }

                : task

        )

    );

}

9. Поиск

const searchedTasks =
    tasks.filter(task =>

        task.title
            .toLowerCase()
            .includes(
                search.toLowerCase()
            )

    );

10. Фильтр

const visibleTasks =
    searchedTasks.filter(task => {

        if (filter === "done") {
            return task.completed;
        }

        if (filter === "active") {
            return !task.completed;
        }

        return true;

    });
