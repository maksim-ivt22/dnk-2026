React №4 — Мини-проект

Собираем полноценное React-приложение

Цель: объединить компоненты, props, useState, map(), filter(), формы, useEffect и fetch в одном проекте.

1. Что будем делать?

Проект называется: 

Study Planner

Это приложение для учебных задач.

Пользователь сможет:

  • видеть список задач;
  • добавлять новые задачи;
  • отмечать выполненные;
  • удалять задачи;
  • искать задачи;
  • фильтровать их;
  • загружать стартовые задачи с API.

Каждая задача — объект.

Все задачи находятся в массиве.

5. Компонент TaskItem

Одна задача — отдельный компонент. 

function TaskItem({
    task,
    toggleTask,
    deleteTask
}) {

    return (

        <div className="task">

            <h3>
                {task.title}
            </h3>

            <p>
                {task.completed
                    ? "Выполнено"
                    : "Не выполнено"}
            </p>

        </div>

    );

}

6. Props

App передаёт данные в TaskItem: 

<TaskItem
    task={task}
    toggleTask={toggleTask}
    deleteTask={deleteTask}
/>

{tasks.map(task => (

8. Добавление задачи

Форма будет отдельным компонентом: 

<TaskForm
    addTask={addTask}
/>

В App: 

function addTask(title) {

    const newTask = {

        id: Date.now(),

        title: title,

        completed: false

    };


    setTasks([
        ...tasks,
        newTask
    ]);

}

9. Удаление

function deleteTask(id) {

    setTasks(
        tasks.filter(
            task => task.id !== id
        )
    );

}

10. Выполнено / не выполнено

function toggleTask(id) {

    setTasks(

        tasks.map(task =>

            task.id === id

                ? {
                    ...task,
                    completed: !task.completed
                }

                : task

        )

    );

}

11. Что здесь происходит?

нашли нужную задачу ↓ создали её копию ↓ completed меняется ↓ setTasks() ↓ React обновляет интерфейс

12. Поиск

const searchedTasks = tasks.filter(

    task =>

        task.title
            .toLowerCase()
            .includes(
                search.toLowerCase()
            )

);

13. Фильтр

const filteredTasks =
    searchedTasks.filter(task => {

        if (filter === "done") {
            return task.completed;
        }

        if (filter === "active") {
            return !task.completed;
        }

        return true;

    });

16. useEffect

useEffect(() => {

    async function loadTasks() {

        const response = await fetch(
            "https://jsonplaceholder.typicode.com/todos?_limit=5"
        );

        const data = await response.json();

        setTasks(data);

    }

    loadTasks();

}, []);
