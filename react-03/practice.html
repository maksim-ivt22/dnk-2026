Создаём приложение «Каталог пользователей»

Задача

Создай React-приложение, которое получает пользователей с внешнего API.

Приложение должно:
  • загрузить пользователей через fetch();
  • хранить их в useState;
  • использовать useEffect;
  • вывести пользователей через map();
  • показать имя;
  • показать email;
  • показать город;
  • иметь состояние загрузки.

13. App.jsx должен выглядеть примерно так

import {
    useEffect,
    useState
} from "react";

import UserCard
    from "./components/UserCard";


function App() {

    const [users, setUsers] = useState([]);

    const [loading, setLoading] = useState(true);

    const [error, setError] = useState("");


    useEffect(() => {

        async function loadUsers() {

            try {

                const response = await fetch(
                    "https://jsonplaceholder.typicode.com/users"
                );


                if (!response.ok) {

                    throw new Error(
                        "Ошибка загрузки"
                    );

                }


                const data =
                    await response.json();


                setUsers(data);

            } catch (error) {

                setError(
                    "Не удалось загрузить пользователей"
                );

            } finally {

                setLoading(false);

            }

        }


        loadUsers();

    }, []);


    return (

        <div className="container">

            <h1>
                Каталог пользователей
            </h1>


            <p>
                Найдено:
                {users.length}
            </p>


            {loading && (

                <p>
                    Загрузка...
                </p>

            )}


            {error && (

                <p>
                    {error}
                </p>

            )}


            <div className="cards">

                {users.map(user => (

                    <UserCard
                        key={user.id}
                        name={user.name}
                        email={user.email}
                        city={
                            user.address.city
                        }
                    />

                ))}

            </div>

        </div>

    );

}


export default App;
